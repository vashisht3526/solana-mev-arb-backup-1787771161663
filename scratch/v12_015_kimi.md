```rust
use std::sync::atomic::{AtomicU8, Ordering};

#[derive(Clone, Copy, Debug, PartialEq)]
pub enum FlightState {
    Idle = 0,
    Validating = 1,      // Pre-flight checks running
    Bundling = 2,        // Jito bundle construction
    InFlight = 3,        // Bundle submitted, awaiting confirmation
    Recovering = 4,      // Failure detected, cleanup in progress
}

pub struct FlightLock {
    state: AtomicU8,
    position_lamports: AtomicU64,
    entry_slot: AtomicU64,  // For timeout-based recovery
}

impl FlightLock {
    pub fn try_acquire(
        &self,
        requested_size: u64,
        wallet_balance: u64,
        current_slot: u64,
    ) -> Result<FlightGuard, FlightError> {
        // CAS transition: Idle -> Validating
        match self.state.compare_exchange(
            FlightState::Idle as u8,
            FlightState::Validating as u8,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(_) => {
                // Validate under lock to prevent race conditions
                let reserve_required = RESERVE_FLOOR_LAMPORTS
                    .saturating_add(POSITION_BUFFER_LAMPORTS)  // 5M buffer
                    .saturating_add(MAX_FEE_ESTIMATE_LAMPORTS); // 10M for fees
                
                let available = wallet_balance.saturating_sub(reserve_required);
                
                if requested_size > available {
                    self.state.store(FlightState::Idle as u8, Ordering::SeqCst);
                    return Err(FlightError::InsufficientBuffer);
                }
                
                // Commit position size atomically
                self.position_lamports.store(requested_size, Ordering::SeqCst);
                self.entry_slot.store(current_slot, Ordering::SeqCst);
                
                // Transition to Bundling
                self.state.store(FlightState::Bundling as u8, Ordering::SeqCst);
                
                Ok(FlightGuard { lock: self })
            }
            Err(current) => Err(FlightError::StateConflict(FlightState::from(current))),
        }
    }
    
    /// Emergency timeout recovery (120 slots ≈ 60 seconds)
    pub fn check_timeout(&self, current_slot: u64) -> bool {
        let entry = self.entry_slot.load(Ordering::SeqCst);
        let state = self.state.load(Ordering::SeqCst);
        
        if state != FlightState::Idle as u8 && current_slot.saturating_sub(entry) > 120 {
            // Force reset - log for manual review
            self.force_reset();
            true
        } else {
            false
        }
    }
}

// RAII guard ensures state cleanup on drop
pub struct FlightGuard<'a> {
    lock: &'a FlightLock,
}

impl Drop for FlightGuard<'_> {
    fn drop(&mut self) {
        self.lock.state.store(FlightState::Idle as u8, Ordering::SeqCst);
        self.lock.position_lamports.store(0, Ordering::SeqCst);
    }
}
```

**Key Improvements:**
1. **State machine prevents partial acquisition races**
2. **120-slot timeout prevents indefinite locks**
3. **RAII guard ensures cleanup even on panic**
4. **5M lamport position buffer + 10M fee estimate**

```rust
pub async fn execute_shotgun(
    jito_client: &JitoClient,
    transactions: &[VersionedTransaction],
    config: &ShotgunConfig,
) -> Result<BundleResult, ShotgunError> {

        submit_with_timeout(jito_client, &config.tier1, bundle1),
        submit_with_timeout(jito_client, &config.tier2, bundle2),
    );
    
    // First success wins, cancel others

            let _ = jito_client.cancel_bundle(&config.tier2.engine_url).await;
            Ok(r1)

            // Cancel tier1 if possible (shouldn't happen with higher tip)
            let _ = jito_client.cancel_bundle(&config.tier1.engine_url).await;
            Ok(r2)
        }
        (Err(e1), Err(e2)) => {
            // Both failed - analyze for retry
            Err(ShotgunError::BothEnginesFailed(e1, e2))
        }
        _ => {
            // Partial state - poll for confirmation
            poll_bundle_status(jito_client, transactions, config.timeout_ms - start.elapsed().as_millis() as u64).await
        }
    }
}
```
